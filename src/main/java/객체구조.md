  +id: Long

  +id: Long
  +title: String

  +endDate: LocalDate
  +status: SessionStatus
  +coverImage: CoverImage
  +enrollmentPolicy: EnrollmentPolicy

  +url: String
  +size: Long
  +type: ImageType
  +width: Integer
  +height: Integer

}

  +value: BigDecimal
}

  +price: Amount
  +maxCapacity: Long
  +isFree(): Boolean
  +remainingSeats(enrolledCount: Long): OptionalLong
}

class Payment {
  +id: Long
  +amount: Amount
  +paymentDate: LocalDateTime
}

class PaymentsModule {
  +completePayment(sessionId: Long, userId: Long, amount: Amount): Payment
}

class RegistrationService {
  +register(session: Session, payment: Payment): Boolean
}

class SessionStatus {
  <<enumeration>>
  PREPARING
  OPEN
  CLOSED
}

  <<enumeration>>
  GIF
  JPG
  PNG
  SVG
}
